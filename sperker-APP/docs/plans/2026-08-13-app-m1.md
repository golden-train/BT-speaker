# App M1（串口连接 + 主控面板）实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 交付可运行的桌面 Web App：通过 Web Serial（Chrome/Edge）连接音箱，实现主控面板（播放控制、音量、状态实时刷新、EQ/音源/电量能力探测置灰）。

**Architecture:** Vue 3 + Pinia 前端，四层结构 `UI → Store → ProtocolClient → Transport`；Transport 抽象支持 WebSerial 与 Mock（测试/模拟器）。协议层严格按 `interface.md` 实现：JSON Lines、串行命令队列、事件驱动、`not_implemented` 能力探测。

**Tech Stack:** Vite 5、Vue 3、TypeScript、Pinia、Vitest、Web Serial API（桌面 Chrome/Edge）。

> **技术栈调整说明**：设计文档原推荐 Tauri 2，但本机无 Rust 工具链；M1 以 Web 前端 + Web Serial 落地（桌面浏览器原生直连串口，无需后端进程），分层架构不变，后续可平滑封装 Tauri/Electron。已同步记录到 `app-design.md` 变更记录。

```ts
export class JsonLineCodec {
  private buf = ''

  /** 追加一个 chunk，返回其中完整的行解析结果（坏行跳过） */
  push(chunk: string): unknown[] {
    this.buf += chunk
    const lines = this.buf.split('\n')
    this.buf = lines.pop() ?? ''
    const out: unknown[] = []
    for (const raw of lines) {
      const line = raw.replace(/\r$/, '')
      if (!line.trim()) continue
      try {
        out.push(JSON.parse(line))
      } catch {
        // 坏行跳过，不中断后续解析
      }
    }
    return out
  }
}
```

import type { Transport } from './transport'

type Step = { in: string; out: string[] }

export class MockTransport implements Transport {
  private cb: ((line: string) => void) | null = null
  private openFlag = false
  private steps: Step[] = []

  onLine(cb: (line: string) => void): void { this.cb = cb }
  isOpen(): boolean { return this.openFlag }

  script(steps: Step[]): void { this.steps = steps }

  async open(): Promise<void> { this.openFlag = true }
  async close(): Promise<void> { this.openFlag = false }

  async write(line: string): Promise<void> {
    const hit = this.steps.find(s => s.in === line)
    const outs = hit ? hit.out : [JSON.stringify({ ok: false, error: 'unknown_command', cmd: line })]
    for (const o of outs) this.cb?.(o)
  }

  /** 测试/模拟辅助：主动推送一行（事件） */
  emit(line: string): void { this.cb?.(line) }
}

```ts
import type { Transport } from '../transport/transport'
import { JsonLineCodec } from './jsonline'
import type { IncomingMsg, ResponseMsg, SpeakerEvt } from './types'

const DEFAULT_TIMEOUT_MS = 2000

interface Pending {
  cmd: string
  resolve: (r: ResponseMsg) => void
  reject: (e: Error) => void
  timer: ReturnType<typeof setTimeout>
}

export class ProtocolClient {
  private codec = new JsonLineCodec()
  private pending: Pending[] = []
  private current: Pending | null = null
  private eventCbs = new Set<(e: SpeakerEvt) => void>()

  constructor(private transport: Transport) {
    this.transport.onLine(line => this.handleLine(line))
  }

  async open(): Promise<void> { await this.transport.open() }
  async close(): Promise<void> { await this.transport.close() }

  /** 逐条发送：入队，等前一条完成后再发 */
  send(req: object, timeoutMs: number = DEFAULT_TIMEOUT_MS): Promise<ResponseMsg> {
    const cmd = JSON.stringify(req)
    return new Promise<ResponseMsg>((resolve, reject) => {
      this.pending.push({ cmd, resolve, reject, timer: setTimeout(() => this.fail(cmd, timeoutMs), timeoutMs) })
      this.pump()
    })
  }

  onEvent(cb: (e: SpeakerEvt) => void): () => void {
    this.eventCbs.add(cb)
    return () => this.eventCbs.delete(cb)
  }

  private pump(): void {
    if (this.current || this.pending.length === 0) return
    this.current = this.pending.shift()!
    void this.transport.write(this.current.cmd)
  }

  private fail(cmd: string, timeoutMs: number): void {
    const i = this.pending.findIndex(p => p.cmd === cmd)
    const p = i >= 0 ? this.pending.splice(i, 1)[0] : this.current
    if (!p) return
    if (this.current === p) this.current = null
    p.reject(new Error(`timeout after ${timeoutMs}ms: ${p.cmd}`))
    this.pump()
  }

  private handleLine(line: string): void {
    for (const msg of this.codec.push(line)) this.route(msg as IncomingMsg)
  }

  private route(msg: IncomingMsg): void {
    if ('evt' in msg) {
      for (const cb of this.eventCbs) cb(msg as SpeakerEvt)
      return
    }
    const p = this.current
    if (!p) return
    clearTimeout(p.timer)
    this.current = null
    p.resolve(msg as ResponseMsg)
    this.pump()
  }
}
```

```ts
import { defineStore } from 'pinia'
import type { ProtocolClient } from '../protocol/protocolClient'
import type { SpeakerEvt, SpeakerStatus, StorageInfo } from '../protocol/types'

export type ConnState = 'disconnected' | 'connecting' | 'connected' | 'error'

interface Capabilities { eq: boolean; source: boolean; battery: boolean }

export const useSpeakerStore = defineStore('speaker', {
  state: () => ({
    conn: 'disconnected' as ConnState,
    status: null as SpeakerStatus | null,
    storage: null as StorageInfo | null,
    fw: '',
    caps: { eq: false, source: false, battery: false } as Capabilities,
    client: null as ProtocolClient | null,
  }),
  getters: {
    connected: s => s.conn === 'connected',
  },
  actions: {
    attach(client: ProtocolClient) {
      this.client = client
      client.onEvent(e => this.onEvent(e))
    },

    async connect() {
      if (!this.client) throw new Error('client not attached')
      this.conn = 'connecting'
      try {
        await this.client.open()
        this.conn = 'connected'
        await this.refresh()
        await this.probe()
      } catch (e) {
        this.conn = 'error'
        throw e
      }
    },

    async disconnect() {
      await this.client?.close()
      this.conn = 'disconnected'
    },

    async refresh() {
      if (!this.client) return
      const st = await this.client.send({ cmd: 'getStatus' })
      if (st.ok && st.status) this.status = st.status
      const sg = await this.client.send({ cmd: 'getStorage' })
      if (sg.ok && sg.storage) this.storage = sg.storage
    },

    async probe() {
      if (!this.client) return
      const checks: Array<keyof Capabilities> = ['eq', 'source', 'battery']
      for (const key of checks) {
        const cmd = key === 'eq' ? 'setEq' : key === 'source' ? 'setSource' : 'getBattery'
        const r = await this.client.send({ cmd })
        const unavailable = r.ok === false && (r.error === 'not_implemented' || r.error === 'unknown_command')
        this.caps[key] = !unavailable
      }
    },

    async setVolume(value: number) {
      await this.client?.send({ cmd: 'setVolume', value })
    },
    async play() { await this.client?.send({ cmd: 'play' }) },
    async pause() { await this.client?.send({ cmd: 'pause' }) },
    async toggle() { await this.client?.send({ cmd: 'toggle' }) },
    async next() { await this.client?.send({ cmd: 'next' }) },
    async prev() { await this.client?.send({ cmd: 'prev' }) },

    onEvent(e: SpeakerEvt) {
      if (!this.status) this.status = { volume: 0, playstate: 'stopped', bt: false, eq: 'flat', source: 'bluetooth', battery: -1, sd: false }
      switch (e.evt) {
        case 'ready': this.fw = e.fw; break
        case 'volume': this.status.volume = e.value; break
        case 'playstate': this.status.playstate = e.state; break
        case 'bt': this.status.bt = e.connected; if (e.name) this.status.btName = e.name; break
        case 'track':
          this.status.title = e.title
          if (e.artist) this.status.artist = e.artist
          break
        case 'mute': this.status.muted = e.muted; break
        case 'error': break // M1 仅记录，UI 弹提示由后续任务处理
      }
    },
  },
})
```

`webSerialTransport.ts`：
```ts
import type { Transport } from './transport'

export class WebSerialTransport implements Transport {
  private port: SerialPort | null = null
  private reader: ReadableStreamDefaultReader<Uint8Array> | null = null
  private decoder = new TextDecoder()
  private lineBuf = ''
  private cb: ((line: string) => void) | null = null
  private openFlag = false

  constructor(port?: SerialPort) { this.port = port ?? null }

  onLine(cb: (line: string) => void): void { this.cb = cb }
  isOpen(): boolean { return this.openFlag }

  async open(): Promise<void> {
    if (!this.port) {
      if (!navigator.serial) throw new Error('浏览器不支持 Web Serial（请用桌面 Chrome/Edge）')
      this.port = await navigator.serial.requestPort()
    }
    await this.port.open({ baudRate: 115200 })
    this.openFlag = true
    void this.readLoop()
  }

  async write(line: string): Promise<void> {
    if (!this.port?.writable) throw new Error('串口不可写')
    const writer = this.port.writable.getWriter()
    await writer.write(new TextEncoder().encode(line + '\n'))
    writer.releaseLock()
  }

  async close(): Promise<void> {
    this.openFlag = false
    try { await this.reader?.cancel() } catch { /* ignore */ }
    if (this.port) { try { await this.port.close() } catch { /* ignore */ } }
    this.port = null
  }

  private async readLoop(): Promise<void> {
    if (!this.port?.readable) return
    const reader = this.port.readable.getReader()
    this.reader = reader
    try {
      while (this.openFlag) {
        const { value, done } = await reader.read()
        if (done) break
        const text = this.decoder.decode(value, { stream: true })
        this.lineBuf += text
        let idx = this.lineBuf.indexOf('\n')
        while (idx >= 0) {
          const line = this.lineBuf.slice(0, idx).replace(/\r$/, '')
          this.lineBuf = this.lineBuf.slice(idx + 1)
          if (line.trim()) this.cb?.(line)
          idx = this.lineBuf.indexOf('\n')
        }
      }
    } catch { /* 串口断开 */ } finally {
      this.openFlag = false
      try { reader.releaseLock() } catch { /* ignore */ }
    }
  }
}
```
